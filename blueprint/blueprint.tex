\author{Juan J. Manfredi \and Aelson Sobral}

\begin{document}
\begin{abstract}
This document collects the natural-language statements and detailed proofs that accompany the Lean~4 formalization of Theorem~1.1 and Lemma~2.1 of J.~J.~Manfredi and A.~Sobral, \emph{$p$-harmonic homogeneous polynomials}.
Each subsection corresponds to the Lean declaration of the same name in the directory \texttt{PHarmonicPoly/}; cross-references point to the results that a proof uses.
Section~\ref{sec:main} lists the formalized results of the paper. Within each section, items appear in dependency order.
The Preamble lists the background facts from Mathlib on which the proofs rely.
\end{abstract}
